{{- /* 子ページ一覧（Confluence の children マクロ）。引数は変換器がマクロの設定から生成する。 - page / space: 一覧の起点にするページのタイトルとスペース（無指定ならこのページ） - all（"true"）/ depth: all が無ければ直下の子だけ。all があれば depth 階層まで（無指定なら上限なし。循環に備えて20階層まで） - sort（creation|title|modified）/ reverse: 無指定なら Confluence のページツリー順（weight） - first: 直下の子の表示件数 - style（h1〜h6）: 直下の子を見出しで表示し、その下の階層を箇条書きにする */ -}} {{- $page := .Page }} {{- $space := .Get "space" | default ($page.Params.space | default "") }} {{- $rootID := $page.Params.page_id | default "" }} {{- with .Get "page" }} {{- $title := . }} {{- $rootID = "" }} {{- range $p := site.RegularPages }} {{- if and (not $rootID) (eq $p.Title $title) (eq ($p.Params.space | default "") $space) }} {{- $rootID = $p.Params.page_id | default "" }} {{- end }} {{- end }} {{- end }} {{- $maxDepth := 1 }} {{- if eq (.Get "all") "true" }} {{- $maxDepth = int (.Get "depth" | default "0") }} {{- if le $maxDepth 0 }}{{ $maxDepth = 20 }}{{ end }} {{- end }} {{- /* 見出しタグ名としてそのまま埋め込むので h1〜h6 以外は受け付けない */}} {{- $style := .Get "style" | default "" }} {{- if not (findRE `^h[1-6]$` $style) }}{{ $style = "" }}{{ end }} {{- /* parent_id → 子ページ の索引（サイドバーと同じく newScratch で線形に組み立てる） */}} {{- $byParentScratch := newScratch }} {{- range $p := site.RegularPages }} {{- with $p.Params.parent_id }}{{ $byParentScratch.Add . (slice $p) }}{{ end }} {{- end }} {{- $byParent := $byParentScratch.Values }} {{- with and $rootID (index $byParent $rootID) }}
